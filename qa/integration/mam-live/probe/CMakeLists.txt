cmake_minimum_required(VERSION 3.16)
project(mam_runtime_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(QT_DEFAULT_MAJOR_VERSION 6 CACHE STRING "Qt major used by the Psi executable")
find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Widgets Test)
add_library(mam_runtime_probe MODULE probe.cpp)
set_target_properties(mam_runtime_probe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/generic")
target_link_libraries(mam_runtime_probe PRIVATE Qt${QT_DEFAULT_MAJOR_VERSION}::Widgets Qt${QT_DEFAULT_MAJOR_VERSION}::Test)
