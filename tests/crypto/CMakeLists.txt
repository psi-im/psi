cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(psi_crypto_regressions LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(QT_DEFAULT_MAJOR_VERSION 6 CACHE STRING "Qt major version")
    enable_testing()
    find_package(Qca3-qt${QT_DEFAULT_MAJOR_VERSION} 3.1.0 CONFIG REQUIRED)
    set(PSI_QCA_TARGET Qca3::Qca)
    list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake/modules")
    include(PsiQCA)
endif()
if(WIN32)
    find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Core Test)
    add_executable(crypto_native_provider_test nativeprovider.cpp)
    set_target_properties(crypto_native_provider_test PROPERTIES AUTOMOC ON)
    target_include_directories(crypto_native_provider_test PRIVATE ../../src)
    target_link_libraries(crypto_native_provider_test Qt${QT_DEFAULT_MAJOR_VERSION}::Test)
    psi_link_qca(crypto_native_provider_test)
    add_test(NAME crypto_native_provider COMMAND crypto_native_provider_test)
    set_tests_properties(crypto_native_provider PROPERTIES TIMEOUT 30)
endif()
