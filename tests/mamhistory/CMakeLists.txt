cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(psi_mam_history LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    enable_testing()
    set(QT_DEFAULT_MAJOR_VERSION 6 CACHE STRING "Qt major version")
endif()
find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Core Sql Test)
add_executable(history_archive_test HistoryArchiveTest.cpp ../../src/historyarchive.cpp)
set_target_properties(history_archive_test PROPERTIES AUTOMOC ON)
target_include_directories(history_archive_test PRIVATE ../../src)
target_link_libraries(history_archive_test PRIVATE Qt${QT_DEFAULT_MAJOR_VERSION}::Sql Qt${QT_DEFAULT_MAJOR_VERSION}::Test)
add_test(NAME history_archive COMMAND history_archive_test)
