cmake_minimum_required(VERSION 3.10)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(psi_filesharing_regressions LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(QT_DEFAULT_MAJOR_VERSION 6 CACHE STRING "Qt major version")
    enable_testing()
endif()
find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Core Test)
add_executable(filesharing_http_test httpranges.cpp ../../src/httputil.cpp)
set_target_properties(filesharing_http_test PROPERTIES AUTOMOC ON)
target_include_directories(filesharing_http_test PRIVATE ../../src)
target_link_libraries(filesharing_http_test PRIVATE Qt${QT_DEFAULT_MAJOR_VERSION}::Core Qt${QT_DEFAULT_MAJOR_VERSION}::Test)
add_test(NAME filesharing_http COMMAND filesharing_http_test)
set_tests_properties(filesharing_http PROPERTIES TIMEOUT 20)
find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Widgets)
add_executable(filesharing_external_player_test externalplayer.cpp ../../src/desktoputil.cpp)
set_target_properties(filesharing_external_player_test PROPERTIES AUTOMOC ON)
target_include_directories(filesharing_external_player_test PRIVATE ../../src)
target_link_libraries(filesharing_external_player_test PRIVATE Qt${QT_DEFAULT_MAJOR_VERSION}::Test
    Qt${QT_DEFAULT_MAJOR_VERSION}::Widgets)
if(WIN32)
    target_link_libraries(filesharing_external_player_test PRIVATE shell32)
endif()
add_test(NAME filesharing_external_player COMMAND filesharing_external_player_test)
set_tests_properties(filesharing_external_player PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
find_program(NODE_EXECUTABLE NAMES node nodejs)
if(NODE_EXECUTABLE)
    add_test(NAME filesharing_media_actions COMMAND ${NODE_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/media-actions.js ${CMAKE_CURRENT_SOURCE_DIR}/../../themes/chatview/util.js)
endif()
