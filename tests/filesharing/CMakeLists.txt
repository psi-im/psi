cmake_minimum_required(VERSION 3.10)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(psi_filesharing_regressions LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(QT_DEFAULT_MAJOR_VERSION 6 CACHE STRING "Qt major version")
    enable_testing()
endif()
find_package(Qt${QT_DEFAULT_MAJOR_VERSION} REQUIRED COMPONENTS Core Test)
add_executable(filesharing_http_test httpranges.cpp ../../src/httputil.cpp)
set_target_properties(filesharing_http_test PROPERTIES AUTOMOC ON)
target_include_directories(filesharing_http_test PRIVATE ../../src)
target_link_libraries(filesharing_http_test PRIVATE Qt${QT_DEFAULT_MAJOR_VERSION}::Core Qt${QT_DEFAULT_MAJOR_VERSION}::Test)
add_test(NAME filesharing_http COMMAND filesharing_http_test)
set_tests_properties(filesharing_http PROPERTIES TIMEOUT 20)
